# Dependencies for an ANALYSIS MODULE — the deployable unit that owns one GPU,
# decodes its assigned camera feeds, detects behaviour, and reports events to the
# central app. This is the heavy stack (torch/CUDA via ultralytics, ~3 GB).
#
# The central app deliberately shares NONE of this: see central/requirements.txt.
# Keeping them apart is what lets central deploy anywhere, without a GPU.

# --- AI / CV stack ---
# 8.4.0 is a HARD floor, not a preference: yolo26*.pt ships in the v8.4.0 asset
# release, so on 8.3.x the auto-download of Config.model_path fails outright.
ultralytics>=8.4.0        # YOLO26 detection + built-in ByteTrack tracking
lap>=0.5.12               # linear assignment, required by the built-in tracker
supervision>=0.25,<0.30   # Detections container + annotators (ByteTrack removed in 0.30)
opencv-python>=4.10.0     # drawing / image ops
av>=12.0                  # PyAV: decode RTSP/HLS/HTTP streams (libav; no system ffmpeg)
numpy>=1.26,<2.2          # array math (kept <2.2 for broad wheel compatibility)
pandas>=2.2.0             # events CSV generation

# --- Web server (module API + video WebSocket to the browser) ---
fastapi>=0.115.0          # routes + WebSocket transport
uvicorn[standard]>=0.30.0 # ASGI server

# --- Reporting to the central app ---
httpx>=0.27.0             # outbound register / heartbeat / event delivery
